cmake_minimum_required(VERSION 3.5)
project(poseEstimation3D)

## The node itself is Python (it drives the C++ engine through ctypes), so the
## only thing that has to be generated here are the three messages — exactly the
## set magician_body_pose_estimation defines, so downstream code only changes
## the package it imports them from.
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Joint3D.msg"
  "msg/Skeleton.msg"
  "msg/Skeletons.msg"
)

## Installed as `poseEstimation3D` so it runs with
##   ros2 run poseEstimation3D poseEstimation3D
## The source file keeps its .py name on purpose: a module named exactly
## `poseEstimation3D` sitting next to the script would shadow the generated
## message package of the same name (the script's own directory comes first on
## sys.path).  The installed copy has no .py suffix, so it is not importable.
install(PROGRAMS poseEstimation3D_node.py
  DESTINATION lib/${PROJECT_NAME}
  RENAME poseEstimation3D
)

## Modules the node imports from its own directory.
install(FILES fsb_engine.py tracker.py
  DESTINATION lib/${PROJECT_NAME}
)

## The installed node sits under <ws>/install, from where the SAM3DBody-cpp
## checkout it drives cannot be found by walking up the directory tree (running
## the script in place can).  Record it now so `ros2 run` needs no --lib-dir.
# Resolve the package directory first: colcon sees it through the workspace
# symlink, and ".." would otherwise be collapsed before the symlink is followed.
get_filename_component(_PKG_REAL "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(_ENGINE_ROOT "${_PKG_REAL}/../../.." REALPATH)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine_root.py"
     "# Generated by CMake — the SAM3DBody-cpp checkout this package was built from.\n"
     "ENGINE_ROOT = r\"${_ENGINE_ROOT}\"\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/engine_root.py"
  DESTINATION lib/${PROJECT_NAME}
)
install(DIRECTORY aruco
  DESTINATION lib/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE
)

ament_export_dependencies(rosidl_default_runtime)
ament_package()
